cmake_minimum_required(VERSION 3.20)
project(svr_review LANGUAGES C CXX ASM_MASM)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
get_filename_component(SVR "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(MH "${SVR}/third_party/minhook")
add_library(review_minhook STATIC ${MH}/src/buffer.c ${MH}/src/hook.c ${MH}/src/trampoline.c ${MH}/src/hde/hde64.c)
target_include_directories(review_minhook PUBLIC ${MH}/include)
add_executable(runtime_review runtime_review.cpp ${SVR}/capture.cpp ${SVR}/compatibility.cpp ${SVR}/fleet_shell.cpp ${SVR}/entry.asm)
target_include_directories(runtime_review PRIVATE ${SVR})
target_compile_definitions(runtime_review PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(runtime_review PRIVATE review_minhook bcrypt)
add_executable(geometry_review geometry_review.cpp ${SVR}/fleet_shell.cpp)
target_include_directories(geometry_review PRIVATE ${SVR})
add_library(failure_minhook STATIC minhook_failure_wrap.c ${MH}/src/buffer.c ${MH}/src/trampoline.c ${MH}/src/hde/hde64.c)
target_include_directories(failure_minhook PRIVATE ${MH}/src PUBLIC ${MH}/include)
add_executable(minhook_failure_review minhook_failure_review.cpp)
target_link_libraries(minhook_failure_review PRIVATE failure_minhook)
