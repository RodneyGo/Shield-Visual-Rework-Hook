cmake_minimum_required(VERSION 3.20)
project(svr_release LANGUAGES C CXX ASM_MASM)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 MSVC required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(MH "${CMAKE_CURRENT_SOURCE_DIR}/third_party/minhook")
add_library(minhook STATIC ${MH}/src/buffer.c ${MH}/src/hook.c ${MH}/src/trampoline.c ${MH}/src/hde/hde64.c)
target_include_directories(minhook PUBLIC ${MH}/include)
add_library(svr_contact_probe SHARED probe.cpp capture.cpp compatibility.cpp abi.cpp fleet_shell.cpp entry.asm)
target_compile_definitions(svr_contact_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(svr_contact_probe PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4>)
target_link_options(svr_contact_probe PRIVATE /PDBALTPATH:svr_contact_probe.pdb)
target_link_libraries(svr_contact_probe PRIVATE minhook bcrypt)
add_executable(probe_selftest selftest.cpp capture.cpp entry.asm)
target_compile_definitions(probe_selftest PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(probe_selftest PRIVATE minhook)
add_executable(loader_selftest loader_selftest.cpp)
add_executable(fleet_direction_selftest fleet_direction_selftest.cpp fleet_shell.cpp)
add_executable(missile_selftest missile_selftest.cpp missile_fixture.asm capture.cpp entry.asm)
target_compile_definitions(missile_selftest PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(missile_selftest PRIVATE minhook)
enable_testing()
add_test(NAME native_capture COMMAND probe_selftest)
add_test(NAME missile_capture COMMAND missile_selftest)

add_executable(compatibility_audit compatibility_audit.cpp compatibility.cpp abi.cpp)
target_compile_definitions(compatibility_audit PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(compatibility_audit PRIVATE bcrypt)

add_executable(runtime_regression tests/runtime_review.cpp capture.cpp compatibility.cpp abi.cpp fleet_shell.cpp entry.asm)
target_include_directories(runtime_regression PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(runtime_regression PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(runtime_regression PRIVATE minhook bcrypt)
add_test(NAME runtime_regression COMMAND runtime_regression)
add_executable(geometry_regression tests/geometry_review.cpp fleet_shell.cpp)
target_include_directories(geometry_regression PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_library(failure_minhook STATIC tests/minhook_failure_wrap.c ${MH}/src/buffer.c ${MH}/src/trampoline.c ${MH}/src/hde/hde64.c)
target_include_directories(failure_minhook PRIVATE ${MH}/src PUBLIC ${MH}/include)
add_executable(minhook_failure_regression tests/minhook_failure_review.cpp)
target_link_libraries(minhook_failure_regression PRIVATE failure_minhook)
foreach(mode RANGE 1 6)
  add_test(NAME minhook_failure_${mode} COMMAND minhook_failure_regression ${mode})
endforeach()

foreach(fixture normal extreme_finite extreme_overflow invalid_index cyclic_node nan_vertex wrong_quanta)
  if(fixture STREQUAL "normal")
    set(expected valid)
  else()
    set(expected invalid)
  endif()
  add_test(NAME geometry_${fixture} COMMAND geometry_regression "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/${fixture}.bin" ${expected})
endforeach()
