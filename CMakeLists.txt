cmake_minimum_required(VERSION 3.20)
project(svr_release LANGUAGES C CXX ASM_MASM)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 MSVC required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(MH "${CMAKE_CURRENT_SOURCE_DIR}/third_party/minhook")
add_library(minhook STATIC ${MH}/src/buffer.c ${MH}/src/hook.c ${MH}/src/trampoline.c ${MH}/src/hde/hde64.c)
target_include_directories(minhook PUBLIC ${MH}/include)
add_library(svr_contact_probe SHARED probe.cpp capture.cpp compatibility.cpp fleet_shell.cpp entry.asm)
target_compile_definitions(svr_contact_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(svr_contact_probe PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4>)
target_link_options(svr_contact_probe PRIVATE /PDBALTPATH:svr_contact_probe.pdb)
target_link_libraries(svr_contact_probe PRIVATE minhook bcrypt)
add_executable(probe_selftest selftest.cpp capture.cpp entry.asm)
target_compile_definitions(probe_selftest PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(probe_selftest PRIVATE minhook)
add_executable(missile_selftest missile_selftest.cpp missile_fixture.asm capture.cpp entry.asm)
target_compile_definitions(missile_selftest PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(missile_selftest PRIVATE minhook)
enable_testing()
add_test(NAME native_capture COMMAND probe_selftest)
add_test(NAME missile_capture COMMAND missile_selftest)

add_executable(compatibility_audit compatibility_audit.cpp compatibility.cpp)
target_compile_definitions(compatibility_audit PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(compatibility_audit PRIVATE bcrypt)
